    function renderContent() {

      const data = SITE_CONTENT[currentLang];


      document.querySelectorAll('[data-i18n]').forEach(el => {

        const key = el.getAttribute('data-i18n');

        if (data[key]) el.textContent = data[key];

      });


      // Метрики

      const metricsEl = document.getElementById('metrics-container');

      metricsEl.innerHTML = data.metrics.map(m => `

        <div class="metric-item">

          <h3>${m.value}</h3>

          <p>${m.label}</p>

        </div>

      `).join('');


      // Услуги / Каналы

      const servicesEl = document.getElementById('services-container');

      const filtered = currentFilter === 'all' 

        ? data.services 

        : data.services.filter(s => s.category === currentFilter);


      servicesEl.innerHTML = filtered.map(s => `

        <div class="service-card">

          <div>

            <div class="service-top">

              <span class="service-badge">${s.badge}</span>

            <h3>${s.title}</h3>

          </div>

          <div class="service-meta">

            <span class="service-geo">${s.geo}</span>

            <button class="service-link" onclick="selectService('${s.title.replace(/'/g, "\\'")}')">${data.card_cta}</button>

          <h4>${c.title}</h4>

          <p>${c.desc}</p>

        </div>

      `).join('');

    }


    function setLang(lang) {

      currentLang = lang;

      document.getElementById('btn-en').classList.toggle('active', lang === 'en');

      document.getElementById('btn-ru').classList.toggle('active', lang === 'ru');

      renderContent();

    }


    document.getElementById('filter-container').addEventListener('click', (e) => {

      if (!e.target.classList.contains('filter-btn')) return;

      document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));

      e.target.classList.add('active');

      currentFilter = e.target.getAttribute('data-filter');

      renderContent();

    });


    function selectService(title) {

      document.getElementById('leadNote').value = `Interested in: ${title}. `;

      const contact = document.getElementById('leadContact').value.trim();
